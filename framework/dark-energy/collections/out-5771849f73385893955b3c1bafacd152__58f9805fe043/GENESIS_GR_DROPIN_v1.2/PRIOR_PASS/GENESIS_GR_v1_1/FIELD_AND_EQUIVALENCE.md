# Field equation and GR equivalence

\[
\boxed{
\operatorname{Ric}^G_{ab}-\frac12\mathcal R_Gg^G_{ab}
+\Lambda_Gg^G_{ab}
=
\kappa_GT^G_{ab}.
}
\]

\[
\boxed{\rho_R^{GR}(\Lambda_G)=\Lambda,\qquad
\rho_R^{GR}(\kappa_G)=8\pi G/c^4.}
\]

\[
\boxed{\nabla_G^aT^G_{ab}=0.}
\]

\[
\rho(g_G)=g,\quad \rho(\nabla_G)=\nabla,\quad
\rho(R_G{}^a{}_{bcd})=R^\rho{}_{\sigma\mu\nu},
\]
\[
\rho(\operatorname{Ric}_G)=R_{\mu\nu},\quad
\rho(\mathcal R_G)=R,\quad
\rho(G_G)=G_{\mu\nu},\quad
\rho(T_G)=T_{\mu\nu}.
\]
Therefore
\[
\boxed{
\rho_R^{GR}\!\left[
G^G_{ab}+\Lambda_Gg^G_{ab}=\kappa_GT^G_{ab}
\right]
=
G_{\mu\nu}+\Lambda g_{\mu\nu}
=
\frac{8\pi G}{c^4}T_{\mu\nu}.
}
\]
