\[
\mathbf{I}_{\lambda}(x) =
\int_{\Sigma}
\phi_{\lambda}(x)\,
\nabla \phi_{\lambda}(x)\,
d\Sigma,
\]

\begin{itemize}
\item $\phi_{\lambda}$ is the eigenfield of spectral band $\lambda$,
\item $\Sigma$ is the manifold slice at fixed recursion depth,
\item $\mathbf{I}_{\lambda}$ measures resistance to spectral deformation.
\end{itemize}
