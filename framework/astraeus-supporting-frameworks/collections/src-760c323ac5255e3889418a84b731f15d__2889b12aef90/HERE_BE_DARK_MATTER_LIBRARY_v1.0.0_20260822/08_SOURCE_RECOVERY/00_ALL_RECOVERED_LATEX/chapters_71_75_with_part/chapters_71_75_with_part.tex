\chapter{Phase-Coherent Civilizations and Collective Stability}

\section{Civilizational Phase Field}
Civilizations carry a macro-phase:
\[
\tau_{\mathrm{civ}} = \sum_i w_i \tau_i.
\]

\section{Stability Criterion}
Stability requires:
\[
(\partial\tau_{\mathrm{civ}})^2 < \Lambda_{\mathrm{crit}}.
\]

\[
\Phi_{\mathrm{flow}} = \int W^{\mathrm{cult}}_{ij}(I_i - I_j)\,dt.
