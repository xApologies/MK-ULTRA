\chapter{Economic Phase Fields and Resource Flow Geometry}

\section{Economic Phase Potential}
Define an economic phase:
\[
\tau_{\mathrm{econ}}(x) = \sum_i w_i R_i(x),
\]
where \(R_i\) represents resource density.

\section{Flow Gradient Law}
Resource flow follows:
\[
J^i = -\kappa\,\partial^i\tau_{\mathrm{econ}}.
\]
